// Замените существующий метод createInterface() в MainActivity.java этим методом.
// Остальные методы и классы MainActivity оставьте без изменений.
private void createInterface() {
    ScrollView scroll = new ScrollView(this);
    scroll.setFillViewport(true);
    scroll.setBackgroundColor(green);

    LinearLayout root = new LinearLayout(this);
    root.setOrientation(LinearLayout.VERTICAL);
    root.setGravity(Gravity.CENTER_HORIZONTAL);
    root.setPadding(dp(18), dp(10), dp(18), dp(20));
    scroll.addView(root);

    TextView title = label("📿  МОЙ ТАСБИХ", gold, 26);
    root.addView(title, params(-1, dp(58)));

    TextView hint = label(
        "Листайте влево или вправо, чтобы сменить зикр",
        cream, 14
    );
    root.addView(hint, params(-1, dp(42)));

    dhikr = label("", Color.WHITE, 28);
    dhikr.setTypeface(null, android.graphics.Typeface.BOLD);
    root.addView(dhikr, params(-1, dp(70)));

    counterPosition = label("", gold, 15);
    root.addView(counterPosition, params(-1, dp(28)));

    // Кнопка создания счётчика занимает прежнее место основной кнопки.
    addAction(root, "＋ Добавить свой счётчик", this::addCounterDialog);

    // Счётчик по центру, кнопки переключения — по бокам.
    LinearLayout counterRow = new LinearLayout(this);
    counterRow.setOrientation(LinearLayout.HORIZONTAL);
    counterRow.setGravity(Gravity.CENTER_VERTICAL);

    Button prev = button("‹");
    prev.setTextSize(32);
    prev.setContentDescription("Назад");
    counterRow.addView(prev, new LinearLayout.LayoutParams(dp(58), dp(62)));

    LinearLayout centerCounter = new LinearLayout(this);
    centerCounter.setOrientation(LinearLayout.VERTICAL);
    centerCounter.setGravity(Gravity.CENTER_HORIZONTAL);

    number = label("0", gold, 68);
    number.setTypeface(null, android.graphics.Typeface.BOLD);
    centerCounter.addView(number, params(-1, dp(96)));

    Button reset = button("Сброс");
    reset.setTextSize(12);
    reset.setMinHeight(0);
    reset.setMinimumHeight(0);
    reset.setPadding(dp(12), 0, dp(12), 0);
    reset.setBackgroundTintList(
        android.content.res.ColorStateList.valueOf(gold)
    );
    centerCounter.addView(reset, new LinearLayout.LayoutParams(
        dp(92), dp(32)
    ));

    LinearLayout.LayoutParams centerParams =
        new LinearLayout.LayoutParams(0, -2, 1);
    centerParams.setMargins(dp(6), 0, dp(6), 0);
    counterRow.addView(centerCounter, centerParams);

    Button next = button("›");
    next.setTextSize(32);
    next.setContentDescription("Вперёд");
    counterRow.addView(next, new LinearLayout.LayoutParams(dp(58), dp(62)));

    root.addView(counterRow, params(-1, dp(132)));

    prev.setOnClickListener(v -> changeCounter(-1));
    next.setOnClickListener(v -> changeCounter(1));

    reset.setOnClickListener(v ->
        new AlertDialog.Builder(this)
            .setTitle("Сбросить счёт?")
            .setMessage("Обнулить этот счётчик?")
            .setNegativeButton("Отмена", null)
            .setPositiveButton("Сброс", (d, w) -> {
                current().count = 0;
                saveCounters();
                update();
            })
            .show()
    );

    goalText = label("", Color.WHITE, 18);
    root.addView(goalText, params(-1, dp(46)));

    // Основная кнопка занимает прежнее место кнопки создания счётчика.
    root.addView(label("ЦЕЛЬ СЧЁТЧИКА", gold, 16),
        params(-1, dp(44)));

    LinearLayout goals = new LinearLayout(this);
    for (int target : new int[]{33, 99, 100}) {
        Button b = button(String.valueOf(target));
        LinearLayout.LayoutParams lp =
            new LinearLayout.LayoutParams(0, dp(58), 1);
        lp.setMargins(dp(4), 0, dp(4), 0);
        goals.addView(b, lp);

        b.setOnClickListener(v -> {
            current().target = target;
            saveCounters();
            update();
        });
    }
    root.addView(goals, params(-1, dp(64)));

    addAction(root, "Своя цель", this::customGoalDialog);
    addAction(root, "✎ Изменить надпись", this::renameCounterDialog);

    addAction(root, "Удалить этот счётчик", () -> {
        if (counters.size() == 1) {
            toast("Нужен хотя бы один счётчик");
            return;
        }

        new AlertDialog.Builder(this)
            .setTitle("Удалить счётчик?")
            .setMessage(current().label)
            .setNegativeButton("Отмена", null)
            .setPositiveButton("Удалить", (d, w) -> {
                counters.remove(active);
                active = Math.min(active, counters.size() - 1);
                saveCounters();
                update();
            })
            .show();
    });

    root.addView(label("НАПОМИНАНИЯ", gold, 18),
        params(-1, dp(46)));

    reminderText = label("", cream, 16);
    root.addView(reminderText, params(-1, dp(65)));

    addAction(root, "＋ Добавить напоминание", this::chooseTime);
    addAction(root, "Изменить / удалить напоминание", this::manageReminders);

    addAction(root, "Отключить все напоминания", () -> {
        for (ReminderItem item : reminders) {
            ReminderReceiver.cancel(this, item.id);
        }

        reminders.clear();
        saveReminders();
        update();
        toast("Напоминания отключены");
    });

    // Кнопка подсчёта расположена внизу экрана, ближе к большому пальцу.
    Button countButton = button("＋  СЧИТАТЬ ЗИКР");
    countButton.setTextSize(24);
    root.addView(countButton, params(-1, dp(88)));

    countButton.setOnClickListener(v -> {
        Counter c = current();
        c.count++;
        saveCounters();
        update();
        vibrate();

        if (c.count == c.target) {
            new AlertDialog.Builder(this)
                .setTitle("Цель достигнута!")
                .setMessage("Альхамдулиллях! Вы выполнили цель.")
                .setPositiveButton("Продолжить", null)
                .show();
        }
    });

    setContentView(scroll);
    update();
}
